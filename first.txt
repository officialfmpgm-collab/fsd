import numpy as np
class Adaline:
    def __init__(self):
        self.weights = np.zeros(3)
    def fit(self, X, y):
        for i in range(10):
            output = np.dot(X, self.weights[1:]) + self.weights[0]
            error = y - output
            self.weights[1:] += 0.01 * np.dot(X.T, error)
            self.weights[0] += 0.01 * np.sum(error)
    def predict(self, X):
        return np.dot(X, self.weights[1:]) + self.weights[0]
X = np.array([[1,2], [2,3], [3,4]])
y = np.array([1,1,0])
a = Adaline()
a.fit(X, y)
print("Weights:", a.weights)
print("Prediction:", a.predict(np.array([2,3])))

3. For a given set of training data examples stored in a .CSV file, implement and demonstrate the Candidate-Elimination algorithm to output a description of the set of all hypotheses consistent with the training examples
import pandas as pd

concepts = np.array(data.iloc[:,0:-1])

for i, h in enumerate(concepts):
    if target[i] == "yes":
        for x in range(len(S)):
            if h[x] != S[x]:
                S[x] = '?'
                G[x][x] = '?'
    if target[i] == "no":
        for x in range(len(S)):
            if h[x] != S[x]:
                G[x][x] = S[x]
            else:
                G[x][x] = '?'
    print("\nStep", i+1)
    print("S =", S)
    print("G =", G)
G = [g for g in G if g != ['?','?','?','?','?','?']]
print("\nFinal Specific Hypothesis:")
print(S)
print("\nFinal General Hypothesis:")
print(G)

4. Write a program to demonstrate the working of the decision tree based ID3algorithm. Use an appropriate data set for building the decision tree and apply this knowledge to classify a new sample.
Code:
import pandas as pd
from sklearn.tree import DecisionTreeClassifier
from sklearn.preprocessing import LabelEncoder
from sklearn.metrics import accuracy_score
data = pd.read_csv(r'C:\Users\sadhv\OneDrive\Desktop\data3.csv')
le = LabelEncoder()
for col in data.columns:
    data[col] = le.fit_transform(data[col])
X = data.iloc[:, :-1]
y = data.iloc[:, -1]
model = DecisionTreeClassifier(criterion='entropy')
model.fit(X, y)
acc = accuracy_score(y, model.predict(X))
test = pd.DataFrame([[0,2,1,0]], columns=X.columns)
pred = model.predict(test)
print("Trained Classifier:", model)
print("\nAccuracy :", acc)
print("\nPrediction :", pred)
if pred[0] == 1:
    print("\nPrediction : Yes")
else:
    print("\nPrediction : No")

 5. Write a program to implement the naïve Bayesian classifier for a sample training data set stored as a .CSV file. Compute the accuracy of the classifier, considering few test data sets.

import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import GaussianNB
from sklearn.metrics import accuracy_score

6.  Assuming a set of documents that need to be classified, use the naïve Bayesian Classifier model to perform this task. Built-in API can be used to write the program. Calculate the accuracy, precision, and recall for your data set.
Code:
import pandas as pd
from sklearn.naive_bayes import MultinomialNB
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics import *
data = pd.read_csv(r'C:\Users\sadhv\OneDrive\Desktop\documents.csv')
X = data['text']
y = data['label']
X = TfidfVectorizer().fit_transform(X)
model = MultinomialNB()
model.fit(X, y)
pred = model.predict(X)
print("Trained Classifier:", model)
print("\nAccuracy :", accuracy_score(y, pred))
print("\nPrecision :", precision_score(y, pred, average='weighted'))
print("\nRecall :", recall_score(y, pred, average='weighted'))
print("\nPrediction :", pred)
print("\nClassification Report:\n")
print(classification_report(y, pred))

P7.  Write a program to construct a Bayesian network considering medical data. Use this model to demonstrate the diagnosis of heart patients using standard  Heart Disease Data Set. You can use Python ML library classes/API.
 
Code:
import pandas as pd
from pgmpy.models import DiscreteBayesianNetwork
from pgmpy.inference import VariableElimination
data = pd.read_csv(r'C:\Users\sadhv\OneDrive\Desktop\heart.csv')
print("Few examples from the dataset are given below\n")
print(data.head())
model = DiscreteBayesianNetwork([
    ('age','heartdisease'),
    ('chol','heartdisease')
])
print("\nLearning CPD using Maximum Likelihood Estimators")
model.fit(data)
infer = VariableElimination(model)
print("\nInferencing with Bayesian Network:")
print("\n1. Probability of HeartDisease given Age=37")
q = infer.query(
    variables=['heartdisease'],
    evidence={'age':37}
)
print(q)
print("\n2. Probability of HeartDisease given Chol=233")
q = infer.query(
    variables=['heartdisease'],
    evidence={'chol':233}
)
print(q)
